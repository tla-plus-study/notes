\section*{Brief Description of the Topics}

Record syntax
\begin{itemize}
\item {
  \begin{verbatim}
  r == [ prof |-> "Fred", num -> 42]
  \end{verbatim}
}
\item {
  record fields are nominal only, position doesn't matter
}
\item {
  \begin{verbatim}
  foo["bar"] == foo.bar
  \end{verbatim}
}
\item {
  \begin{verbatim}
  f EXCEPT !["prof"] = "RED" == f EXCEPT !.prof = "RED"
  \end{verbatim}
  \[
  f'(x) = \begin{cases}
    f'(x) = f(x) & \forall x \in dom(f) \setminus \{"prof"\} \\
    f'(x) = RED  & x = "prof"
  \end{cases}
  \]
}
\end{itemize}
